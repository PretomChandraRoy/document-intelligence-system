1"""Stage 3: turn each invoice's text into fields, choosing between the tiers. 

2 

3Decision for each invoice: 

4 

51. Run the rules. If every required field was found and all the 

6 arithmetic agrees, accept: method "rules", no model call. 

72. Otherwise ask the model. Then merge field by field: 

8 - both agree -> keep, high confidence 

9 - only one has a value -> keep it, with that source's confidence 

10 - they disagree -> for the money fields, keep whichever set makes 

11 the arithmetic agree; otherwise take the 

12 model's value with low confidence, so a person 

13 checks it 

143. If the model is unavailable, the caller retries the task; after the 

15 last retry the rules result is used and marked low confidence. 

16 

17For scanned pages, confidence is capped by the OCR confidence of the page, 

18so a badly read scan always goes to review. 

19"""

34def _eq(a, b) -> bool: 

35 if a is None or b is None: 

36 return False 

37 if isinstance(a, str) and isinstance(b, str): 

38 return " ".join(a.lower().split()) == " ".join(b.lower().split()) 

39 return a == b

42def _lines_ok(inv: ExtractedInvoice) -> bool: 

43 if not inv.line_items or inv.subtotal is None: 

44 return False 

45 if any(li.amount is None for li in inv.line_items): 

46 return False 

47 return abs(sum(li.amount for li in inv.line_items) - inv.subtotal) <= Decimal("0.01")

50def extract_invoice(text: str, client: LLMClient | None, *, page_texts: list[str] | None = None, 

51 ocr_confidence: float | None = None, force_llm: bool = False, 

52 llm_only: bool = False) -> ExtractedInvoice: 

53 rules = extract_rules(text) 

54 if llm_only: 

55 result = llm_extract(text, client, page_texts) 

56 elif not force_llm and not rules.missing() and arithmetic_consistent(rules): 

57 result = rules 

58 elif client is None: 

59 result = rules 

60 for f in list(result.confidence): 

61 result.confidence[f] = min(result.confidence[f], 0.5) 

62 result.warnings.append({"code": "low_confidence", "field": None, 

63 "message": "Language model unavailable; rules-only result needs checking"}) 

64 else: 

65 model = llm_extract(text, client, page_texts) 

66 result = merge(rules, model) 

67 

68 if ocr_confidence is not None: 

69 # On scans the supplier is usually guessed from the first readable line, 

70 # and the evaluation showed that guess is wrong often enough (for example 

71 # when the name sits on a dark band OCR cannot read) to be the main source 

72 # of silent errors. A guessed supplier on OCR text is therefore always reviewed. 

73 for f in rules.guessed: 

74 if f in result.confidence and getattr(result, f) == getattr(rules, f): 

75 result.confidence[f] = min(result.confidence[f], 0.55) 

76 cap = max(0.3, min(1.0, ocr_confidence / 100 + 0.1)) 

77 for f in list(result.confidence): 

78 result.confidence[f] = round(min(result.confidence[f], cap), 3) 

79 return result

82def merge(rules: ExtractedInvoice, model: ExtractedInvoice) -> ExtractedInvoice: 

83 out = copy.deepcopy(rules) 

84 out.raw_model_output = model.raw_model_output 

85 out.warnings = rules.warnings + model.warnings 

86 used_model = False 

87 conflicts = {} 

88 

89 for f in HEADER_FIELDS: 

90 r, m = getattr(rules, f), getattr(model, f) 

91 if r is None and m is None: 

92 continue 

93 if _eq(r, m): 

94 out.confidence[f] = AGREE 

95 elif r is None: 

96 setattr(out, f, m) 

97 out.confidence[f] = model.confidence.get(f, 0.8) 

98 out.evidence[f] = model.evidence.get(f, "") 

99 used_model = True 

100 elif m is None: 

101 pass # keep the rules value and its confidence 

102 else: 

103 conflicts[f] = (r, m) 

104 

105 # money conflicts: prefer the reading that makes the sums agree 

106 money_conf = {f for f in conflicts if f in MONEY_FIELDS} 

107 if money_conf: 

108 trial = copy.deepcopy(out) 

109 for f in money_conf: 

110 setattr(trial, f, conflicts[f][1]) 

111 rules_ok, model_ok = arithmetic_consistent(out), arithmetic_consistent(trial) 

112 for f in money_conf: 

113 if model_ok and not rules_ok: 

114 setattr(out, f, conflicts[f][1]) 

115 out.confidence[f] = 0.85 

116 out.evidence[f] = model.evidence.get(f, "") 

117 used_model = True 

118 elif rules_ok and not model_ok: 

119 out.confidence[f] = 0.85 

120 else: 

121 _flag_conflict(out, f, *conflicts[f], model) 

122 used_model = True 

123 for f in set(conflicts) - money_conf: 

124 _flag_conflict(out, f, *conflicts[f], model) 

125 used_model = True 

126 

127 # line items: keep the list whose total matches the subtotal 

128 if not _lines_ok(rules) and model.line_items: 

129 trial = copy.deepcopy(out) 

130 trial.line_items = model.line_items 

131 if _lines_ok(trial) or len(model.line_items) > len(rules.line_items): 

132 out.line_items = model.line_items 

133 out.confidence["line_items"] = 0.85 if _lines_ok(trial) else DISAGREE 

134 used_model = True 

135 elif rules.line_items and model.line_items and len(rules.line_items) == len(model.line_items): 

136 out.confidence["line_items"] = AGREE 

137 

138 out.method = "llm" if used_model else "rules" 

139 return out

142def _flag_conflict(out: ExtractedInvoice, f: str, r, m, model: ExtractedInvoice) -> None: 

143 setattr(out, f, m) 

144 out.confidence[f] = DISAGREE 

145 out.evidence[f] = model.evidence.get(f, "") 

146 out.warnings.append({"code": "low_confidence", "field": f, 

147 "message": f"Rules read '{r}' but the model read '{m}'; please check", 

148 "details": {"rules": str(r), "model": str(m)}})

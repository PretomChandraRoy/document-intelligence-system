21@router.get("/search", response_model=SearchResponse) 

22@limiter.limit(lambda: get_settings().rate_limit_search) 

23def search(request: Request, q: str | None = Query(None, max_length=500), 

24 mode: str = Query("hybrid", pattern="^(keyword|semantic|hybrid)$"), 

25 limit: int = Query(20, ge=1, le=100), f: Filters = Depends(), db: Session = Depends(get_db)): 

26 base = f.apply(select(Invoice)) 

27 if not q or not q.strip(): 

28 rows = list(db.scalars(base.order_by(Invoice.invoice_date.desc().nullslast()).limit(limit))) 

29 return SearchResponse(mode=mode, query=q, results=[_hit(r, 0.0, None, None, None) for r in rows]) 

30 

31 keyword: list[tuple] = [] 

32 semantic: list[tuple] = [] 

33 if mode in ("keyword", "hybrid"): 

34 tsq = func.websearch_to_tsquery("simple", q) 

35 rank = func.ts_rank(Invoice.search_tsv, tsq).label("rank") 

36 sub = base.subquery() 

37 rows = db.execute(select(Invoice, rank).join(sub, sub.c.id == Invoice.id) 

38 .where(Invoice.search_tsv.op("@@")(tsq)).order_by(rank.desc()).limit(limit * 3)).all() 

39 # also match partial names and numbers that full-text search splits differently 

40 if not rows: 

41 like = f"%{q.strip()}%" 

42 rows = [(r, 0.01) for r in db.scalars(base.where( 

43 Invoice.supplier_name.ilike(like) | Invoice.customer_name.ilike(like) | 

44 Invoice.invoice_number.ilike(like)).limit(limit * 3))] 

45 keyword = [(r, float(s)) for r, s in rows] 

46 if mode in ("semantic", "hybrid"): 

47 try: 

48 client = get_client() 

49 vec = client.embed([q])[0] 

50 dist = Invoice.embedding.cosine_distance(vec).label("dist") 

51 sub = base.subquery() 

52 rows = db.execute(select(Invoice, dist).join(sub, sub.c.id == Invoice.id) 

53 .where(Invoice.embedding.is_not(None)).order_by(dist).limit(limit * 3)).all() 

54 floor = getattr(client, "min_similarity", get_settings().semantic_min_similarity) 

55 semantic = [(r, 1 - float(d)) for r, d in rows if 1 - float(d) >= floor] 

56 except Exception: 

57 semantic = [] # model unavailable: keyword results still returned 

58 

59 scores: dict = {} 

60 for ranklist, key in ((keyword, "k"), (semantic, "s")): 

61 for pos, (inv, _) in enumerate(ranklist, start=1): 

62 e = scores.setdefault(inv.id, {"inv": inv, "score": 0.0, "k": None, "s": None}) 

63 e["score"] += 1.0 / (RRF_K + pos) 

64 e[key] = pos 

65 ranked = sorted(scores.values(), key=lambda e: -e["score"])[:limit] 

66 snippets = _snippets(db, q, [e["inv"].id for e in ranked]) if keyword else {} 

67 return SearchResponse(mode=mode, query=q, results=[ 

68 _hit(e["inv"], round(e["score"], 5), e["k"], e["s"], snippets.get(e["inv"].id)) for e in ranked])

71def _snippets(db: Session, q: str, ids) -> dict: 

72 if not ids: 

73 return {} 

74 tsq = func.websearch_to_tsquery("simple", q) 

75 rows = db.execute(select(Invoice.id, func.ts_headline( 

76 "simple", Invoice.search_text, tsq, 

77 literal_column("'MaxFragments=1,MaxWords=18,MinWords=6,StartSel=<b>,StopSel=</b>'"))) 

78 .where(Invoice.id.in_(ids))).all() 

79 return dict(rows)

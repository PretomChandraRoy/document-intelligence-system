1"""OCR for scanned pages and images: clean the image, then run Tesseract. 

2 

3Tesseract is trained on clean, upright, high-resolution black-on-white text. 

4Each step below moves a real scan closer to that: 

5 

61. upscale low-resolution scans to about 300 dpi, the size Tesseract expects 

72. flatten uneven lighting by dividing by an estimate of the background 

83. deskew by measuring the angle of the text lines and rotating back 

94. denoise with a small median filter, which removes specks but keeps edges 

105. binarise with Otsu's method, which picks the black/white threshold 

11 from the image's own histogram 

12""" 

13

37def preprocess(img: Image.Image, target_width: int = 2000, binarise: bool = False) -> np.ndarray: 

38 a = np.array(img.convert("L")) 

39 

40 # denoise first, at the original size: noise grows when the image is enlarged 

41 a = cv2.fastNlMeansDenoising(a, None, h=12, templateWindowSize=7, searchWindowSize=21) \ 

42 if a.shape[1] <= 1400 else cv2.medianBlur(a, 3) 

43 

44 # deskew at the original size as well, where it is cheaper 

45 a = _deskew(a) 

46 

47 # upscale small images so letters are large enough for Tesseract 

48 h, w = a.shape 

49 if w < target_width * 0.85: 

50 scale = target_width / w 

51 a = cv2.resize(a, (int(w * scale), int(h * scale)), interpolation=cv2.INTER_CUBIC) 

52 

53 # flatten uneven lighting: divide by a heavily blurred copy (the paper colour) 

54 small = cv2.resize(a, None, fx=0.125, fy=0.125) 

55 bg = cv2.resize(cv2.medianBlur(small, 21), (a.shape[1], a.shape[0]), interpolation=cv2.INTER_LINEAR) 

56 a = cv2.divide(a, bg, scale=255) 

57 

58 if binarise: 

59 _, a = cv2.threshold(a, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU) 

60 return a

63def _deskew(a: np.ndarray, max_angle: float = 6.0, step: float = 0.25) -> np.ndarray: 

64 """Find the rotation that makes text lines most horizontal. 

65 

66 For each candidate angle the dark pixels are summed row by row. When the 

67 lines are level, rows alternate sharply between ink and blank paper, so 

68 the variance of those row sums is highest at the correct angle. 

69 """ 

70 small = cv2.resize(a, None, fx=600 / a.shape[1], fy=600 / a.shape[1]) if a.shape[1] > 600 else a 

71 inv = cv2.threshold(small, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)[1].astype(np.float32) 

72 h, w = inv.shape 

73 best, best_score = 0.0, -1.0 

74 for ang in np.arange(-max_angle, max_angle + step, step): 

75 m = cv2.getRotationMatrix2D((w / 2, h / 2), ang, 1.0) 

76 rot = cv2.warpAffine(inv, m, (w, h), flags=cv2.INTER_NEAREST, borderValue=0) 

77 score = float(np.var(rot.sum(axis=1))) 

78 if score > best_score: 

79 best, best_score = float(ang), score 

80 if abs(best) < 0.2: 

81 return a 

82 H, W = a.shape 

83 m = cv2.getRotationMatrix2D((W / 2, H / 2), best, 1.0) 

84 return cv2.warpAffine(a, m, (W, H), flags=cv2.INTER_CUBIC, borderValue=255)

87def ocr_image(img: Image.Image, lang: str = "eng", psm: int = 4, clean: bool = True) -> OcrResult: 

88 t0 = time.perf_counter() 

89 arr = preprocess(img) if clean else np.array(img.convert("L")) 

90 data = pytesseract.image_to_data(arr, lang=lang, config=f"--oem 1 --psm {psm}", 

91 output_type=pytesseract.Output.DICT) 

92 words, confs = [], [] 

93 for i, txt in enumerate(data["text"]): 

94 txt = txt.strip() 

95 conf = float(data["conf"][i]) 

96 if not txt or conf < 0: 

97 continue 

98 x, y, w, h = data["left"][i], data["top"][i], data["width"][i], data["height"][i] 

99 words.append(Word(x, y, x + w, y + h, txt, conf)) 

100 confs.append(conf) 

101 text = "\n".join(lines_from_words(words)) 

102 return OcrResult(text, float(np.mean(confs)) if confs else 0.0, words, 

103 int((time.perf_counter() - t0) * 1000))

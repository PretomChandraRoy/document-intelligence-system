57def norm_label(s) -> str: 

58 s = re.sub(r"\(.*?\)", " ", str(s or "").lower()) # drop "(15%)" 

59 s = s.replace("#", " no ").replace("&", " and ") 

60 s = re.sub(r"[^a-z0-9 ]", " ", s) 

61 return re.sub(r"\s+", " ", s).strip()

64def match_label(text, vocab: dict[str, list[str]], exact_only: bool = False) -> str | None: 

65 """Return the field whose label matches `text`, preferring the longest match.""" 

66 t = norm_label(text) 

67 if not t: 

68 return None 

69 best, best_len = None, 0 

70 for fieldname, labels in vocab.items(): 

71 for lab in labels: 

72 if t == lab and len(lab) > best_len: 

73 best, best_len = fieldname, len(lab) + 1000 # exact beats prefix 

74 elif not exact_only and (t.startswith(lab + " ")) and len(lab) > best_len and best_len < 1000: 

75 best, best_len = fieldname, len(lab) 

76 return best

79def tax_rate_in(text: str): 

80 """'VAT (15%)' -> Decimal('15').""" 

81 from decimal import Decimal 

82 m = re.search(r"(\d{1,2}(?:[.,]\d{1,3})?)\s*%", str(text or "")) 

83 return Decimal(m.group(1).replace(",", ".")) if m else None

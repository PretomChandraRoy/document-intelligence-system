33@router.post("", status_code=202, response_model=UploadResponse) 

34@limiter.limit(lambda: get_settings().rate_limit_upload) 

35def upload(request: Request, files: list[UploadFile] = File(...), db: Session = Depends(get_db)): 

36 s = get_settings() 

37 if len(files) > s.max_files_per_request: 

38 raise HTTPException(413, detail={"code": "too_many_files", 

39 "message": f"At most {s.max_files_per_request} files per request"}) 

40 from ..worker import enqueue 

41 accepted, rejected, to_queue = [], [], [] 

42 for f in files: 

43 name = (f.filename or "unnamed")[:255] 

44 ext = extension_of(name) 

45 try: 

46 stored = storage.save_stream(f.file, s.max_file_mb * 1024 * 1024) 

47 except storage.FileTooLarge: 

48 rejected.append(UploadRejected(filename=name, code="file_too_large", 

49 message=f"Files may be at most {s.max_file_mb} MB")) 

50 continue 

51 detected = sniff(stored.head, storage.absolute(stored.relative_path)) 

52 if ext not in ALLOWED_EXTENSIONS and detected in (None, "empty"): 

53 rejected.append(UploadRejected(filename=name, code="unsupported_format", 

54 message="Accepted formats: PDF, JPG, PNG, TIFF, XLSX, XLS, CSV")) 

55 continue 

56 # the declared format is the extension if it is an allowed one; the 

57 # ingest stage then fails the document if the content disagrees 

58 fmt = ext if ext in ALLOWED_EXTENSIONS else detected 

59 dup = db.scalar(select(Document.id).where(Document.sha256 == stored.sha256).order_by(Document.uploaded_at)) 

60 doc = Document(original_filename=name, sha256=stored.sha256, format=fmt, size_bytes=stored.size_bytes, 

61 storage_path=stored.relative_path, exact_duplicate_of=dup) 

62 db.add(doc) 

63 db.flush() 

64 run = services.start_run(db, doc, "upload") 

65 to_queue.append((str(doc.id), str(run.id))) 

66 accepted.append(UploadAccepted(id=doc.id, filename=name, format=fmt, status=doc.status, 

67 exact_duplicate_of=dup)) 

68 db.commit() 

69 for doc_id, run_id in to_queue: 

70 enqueue(doc_id, run_id) 

71 return UploadResponse(accepted=accepted, rejected=rejected)

74def _brief(db: Session, docs: list[Document]) -> list[dict]: 

75 ids = [d.id for d in docs] 

76 counts = dict(db.execute(select(Invoice.document_id, func.count()).where(Invoice.document_id.in_(ids)) 

77 .group_by(Invoice.document_id)).all()) if ids else {} 

78 out = [] 

79 for d in docs: 

80 b = DocumentBrief.model_validate(d) 

81 b.invoice_count = counts.get(d.id, 0) 

82 out.append(b.model_dump(by_alias=True)) 

83 return out

86@router.get("") 

87def list_documents(status: str | None = None, format: str | None = None, page: int = Query(1, ge=1), 

88 page_size: int = Query(25, ge=1, le=100), db: Session = Depends(get_db)): 

89 q = select(Document) 

90 if status: 

91 q = q.where(Document.status == status) 

92 if format: 

93 q = q.where(Document.format == format) 

94 total = db.scalar(select(func.count()).select_from(q.subquery())) 

95 docs = list(db.scalars(q.order_by(Document.uploaded_at.desc()).offset((page - 1) * page_size).limit(page_size))) 

96 return {"items": _brief(db, docs), "page": page, "page_size": page_size, "total": total}

99@router.get("/{doc_id}") 

100def get_document(doc_id: uuid.UUID, db: Session = Depends(get_db)): 

101 doc = db.scalar(select(Document).where(Document.id == doc_id) 

102 .options(selectinload(Document.pages), selectinload(Document.invoices), 

103 selectinload(Document.runs))) 

104 if not doc: 

105 raise not_found() 

106 counts = dict(db.execute(select(ValidationIssue.invoice_id, func.count()) 

107 .where(ValidationIssue.document_id == doc.id, ValidationIssue.resolved.is_(False)) 

108 .group_by(ValidationIssue.invoice_id)).all()) 

109 invoices = [] 

110 for inv in doc.invoices: 

111 b = InvoiceBrief.model_validate(inv) 

112 b.issue_count = counts.get(inv.id, 0) 

113 invoices.append(b) 

114 detail = DocumentDetail.model_validate(doc) 

115 detail.invoices = invoices 

116 detail.invoice_count = len(invoices) 

117 detail.last_run = RunOut.model_validate(doc.runs[-1]) if doc.runs else None 

118 return detail.model_dump(by_alias=True, mode="json")

140@router.get("/{doc_id}/pages/{n}/image") 

141def page_image(doc_id: uuid.UUID, n: int, dpi: int = Query(110, ge=50, le=200), db: Session = Depends(get_db)): 

142 doc = db.get(Document, doc_id) 

143 if not doc: 

144 raise not_found() 

145 if doc.format not in ("pdf", "jpg", "png", "tiff") or not doc.page_count or not 1 <= n <= doc.page_count: 

146 raise not_found("Page") 

147 from ..pipeline.ingest import render_image_page, render_pdf_page 

148 try: 

149 im = render_pdf_page(storage.absolute(doc.storage_path), n, dpi) if doc.format == "pdf" \ 

150 else render_image_page(storage.absolute(doc.storage_path), n) 

151 except Exception: 

152 raise not_found("Page") 

153 im.thumbnail((1400, 1980)) 

154 buf = io.BytesIO() 

155 im.save(buf, "PNG", optimize=True) 

156 return Response(buf.getvalue(), media_type="image/png", headers={"Cache-Control": "private, max-age=3600"})

160def reprocess_failed(body: BulkReprocessRequest | None = None, db: Session = Depends(get_db)): 

161 from ..worker import enqueue

175@router.post("/{doc_id}/reprocess", status_code=202, response_model=RunOut) 

176def reprocess(doc_id: uuid.UUID, body: ReprocessRequest | None = None, db: Session = Depends(get_db)): 

177 from ..worker import enqueue 

178 doc = db.get(Document, doc_id) 

179 if not doc: 

180 raise not_found() 

181 if doc.status in ("queued", "processing"): 

182 raise HTTPException(409, detail={"code": "busy", "message": "The document is still being processed"}) 

183 stage = (body.from_stage if body else "ingest") 

184 if stage != "ingest" and not doc.pages: 

185 stage = "ingest" # nothing to reuse 

186 run = services.start_run(db, doc, "reprocess", stage) 

187 db.commit() 

188 enqueue(str(doc.id), str(run.id), stage) 

189 db.refresh(run) 

190 return run
